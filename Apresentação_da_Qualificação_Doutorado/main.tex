\begin{frame}{Formulação do Problema}
    \footnotesize
    O objetivo do agente é \textbf{maximizar a taxa de tarefas concluídas dentro do prazo}, minimizando latência e energia.
        
    \begin{equation}
        \max_{\pi}\; J(\pi) = \mathbb{E}_{\pi}\!\left[ w_s \cdot \underbrace{\mathbb{1}(\widehat{\text{JCT}}_t \leq D_t)}_{\text{taxa de sucesso}} - w_c \cdot \underbrace{\frac{1}{2}\!\left(\frac{\widehat{\text{JCT}}_t}{\text{JCT}^{\max}_t} + \frac{E_t}{E^{\max}_t}\right)}_{\text{custo normalizado}\;\in\,[0, 1]} \right]
        \label{eq:obj_geral}
    \end{equation}

    \begin{equation}
        \text{JCT}^{\max}_t = \max\!\bigl(\widehat{\text{JCT}}^{\ell}_t,\; \widehat{\text{JCT}}^{\text{vec}}_t \cdot \mathbb{1}_{\text{viável}},\; \widehat{\text{JCT}}^{\text{v2v}}_t \cdot \mathbb{1}_{\text{viável}}\bigr)
        \label{eq:jct_max}
    \end{equation}

    \begin{equation}
        E^{\text{max}}_t = \max\!\bigl(E^{\ell}_t,\; E^{\text{vec}}_t \cdot \mathbb{1}_{\text{viável}},\; E^{\text{v2v}}_t \cdot \mathbb{1}_{\text{viável}}\bigr)
        \label{eq:e_max}
    \end{equation}

\end{frame}

                \begin{equation*}
                    J(\pi) = \mathbb{E} \left[ \sum_{t} r(s_t, a_t) + \alpha \mathcal{H}(\pi(\cdot|s_t)) \right]
                \end{equation*}
                \vspace{2pt}
                $\bullet$ Incentiva o agente a manter alternativas ativas, impedindo a convergência prematura para ótimos locais.\\
                $\bullet$ A política estocástica se reajusta autonomamente a flutuações rápidas de carga sem regras estáticas rígidas.

\begin{frame}{Design da Função de Recompensa}
    \vspace*{0.2cm}
    \begin{center}
        \begin{equation*}
            r_{t} =
            \begin{cases}
                1{,}0 - \frac{1}{4} \left( \frac{\text{JCT}_t}{D_t} + \frac{E_t}{E^{\max}_t} \right), & \text{se } \text{JCT}_t \leq D_t \text{ \scriptsize (Sucesso)} \\[8pt]
                -1{,}0 - \min \left( 1{,}0, \frac{\text{JCT}_t - D_t}{D_t} \right), & \text{se } \text{JCT}_t > D_t \text{ \scriptsize (Falha)}
            \end{cases}
        \end{equation*}        
    \end{center}

    \vspace{0.1cm}
    \begin{tcolorbox}[colback=blue!5, colframe=blue!80!black, title=Fundamentação do Design, fonttitle=\bfseries\footnotesize, arc=4pt]
        \footnotesize
        \begin{itemize}\setlength\itemsep{0.3em}
            \item \textbf{Penalidade QoS (Falha):} Incorpora restrições críticas aplicando multas severas e progressivas caso a tarefa viole o tempo limite estipulado~\cite{miriyala2026}.
            \item \textbf{Otimização Multiobjetivo (Sucesso):} Uma vez garantido o prazo, a função foca em equilibrar de forma adaptativa a minimização da latência e do consumo de energia~\cite{xue2025}.
            \item \textbf{Estabilidade de Gradiente (Normalização):} A divisão pelos limites máximos ($D_t$ e $E^{\max}_t$) evita que variáveis com grandezas distintas (Segundos vs. Joules) afetem desproporcionalmente a otimização da rede neural~\cite{xue2025}.
        \end{itemize}
    \end{tcolorbox}
\end{frame}

\section{Conclusões e Cronograma}
%%################## SLIDE: CONCLUSÕES ##################
\begin{frame}{Conclusões}
    \begin{itemize}
        \item \textbf{Arquitetura SAC-FTOPSIS}: Integração eficaz entre ranqueamento multicritério (TOPSIS) e Aprendizado por Reforço (SAC).
        \item \textbf{Escalabilidade}: O estado compacto 6D resolve o problema de número variável de veículos candidatos, garantindo uma interface invariante.
        \item \textbf{Contribuições Técnicas}:
            \begin{itemize}
                \scriptsize
                \item Engenharia de recompensa \textit{deadline-aware} com foco em prazos rigorosos.
                \item Redução de 8,6x no JCT médio em comparação ao SAC-Base.
                \item Superação da heurística oráculo (\textit{Greedy-Unfair}) em cenários reais.
            \end{itemize}
        \item \textbf{Métrica SWQ}: Nova métrica que pondera sucesso e folga temporal, provando-se superior à taxa de sucesso simples para avaliação de QoS.
    \end{itemize}
    \begin{tcolorbox}[colback=blue!5, colframe=blue!40, arc=4pt, boxrule=0.5pt]
        \scriptsize \textbf{Resumo}: A proposta demonstrou ser uma solução leve e adaptativa, capaz de operar em ambientes veiculares dinâmicos sem necessidade de retreinamento.
    \end{tcolorbox}
\end{frame}
